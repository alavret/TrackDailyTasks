# ================================================================================
#                     РАСПИСАНИЕ ЗАПУСКА МОДУЛЕЙ track_tasks.py
# ================================================================================
#
# Этот файл используется скриптом track_tasks.py в режиме --auto для определения
# расписания автоматического запуска модулей.
#
# Запуск в режиме автоматического выполнения:
#     python track_tasks.py --auto
#
# Файл автоматически перечитывается каждые CRON_REFRESH минут (по умолчанию 5).
# Это позволяет изменять расписание без перезапуска скрипта.
#
# Примечание: список доступных модулей загружается только при запуске скрипта.
# Для добавления новых модулей требуется перезапуск скрипта.
# Запускаемые модули должны быть указаны в RUN_MODULES в файле .env.
#
# ================================================================================
#                          ФОРМАТ CRON
# ================================================================================
#
# Формат расписания состоит из 5 полей + имена модулей:
#
#     *    *    *    *    *    модули
#     │    │    │    │    │       │
#     │    │    │    │    │       └── Модули через запятую
#     │    │    │    │    └────────── День недели (0-7 или SUN-SAT)
#     │    │    │    └─────────────── Месяц (1-12 или JAN-DEC)
#     │    │    └──────────────────── День месяца (1-31)
#     │    └───────────────────────── Час (0-23)
#     └────────────────────────────── Минута (0-59)
#
# ================================================================================
#                          СПЕЦИАЛЬНЫЕ СИМВОЛЫ
# ================================================================================
#
# *       Любое значение
# ?       Любое значение (для day-of-month/day-of-week, когда другое поле задано)
# ,       Список значений (например: MON,WED,FRI)
# -       Диапазон значений (например: MON-FRI, 1-15)
# /       Шаг (например: 0/15 - каждые 15 единиц начиная с 0)
# L       Последний день (L - последний день месяца, 6L - последняя суббота)
# W       Ближайший рабочий день (15W - ближайший рабочий день к 15-му)
# #       N-й день недели в месяце (6#3 - третья суббота месяца)
#
# ================================================================================
#                          ДНИ НЕДЕЛИ
# ================================================================================
#
# Английские аббревиатуры:
#     SUN (0), MON (1), TUE (2), WED (3), THU (4), FRI (5), SAT (6)
#
# Русские аббревиатуры (поддерживаются):
#     ВС (0), ПН (1), ВТ (2), СР (3), ЧТ (4), ПТ (5), СБ (6)
#
# Примечание: 0 и 7 оба означают воскресенье
#
# ================================================================================
#                          МОДУЛИ
# ================================================================================
#
# Модули располагаются в каталоге скрипта и имеют имена вида module_<name>.py
# В расписании указывается только <name>, без префикса module_ и расширения .py
#
# Доступные модули определяются автоматически при запуске скрипта.
# Каждый модуль должен содержать функцию run(settings).
#
# Несколько модулей указываются через запятую без пробелов:
#     delete_users,license
#
# ================================================================================
#                          СИНТАКСИС СТРОК РАСПИСАНИЯ
# ================================================================================
#
# Формат строки:
#     <мин> <час> <день_месяца> <месяц> <день_недели> <модули>
#
# Комментарии начинаются с символа # и игнорируются.
# Пустые строки игнорируются.
#
# ================================================================================
#                          ПРИМЕРЫ РАСПИСАНИЙ
# ================================================================================
#
# Каждые 3 часа с понедельника по пятницу (в 0, 3, 6, 9, 12, 15, 18, 21 часов)
# 0 0/3 ? * MON-FRI delete_users,license
#
# Ежедневно в 10:15
# 15 10 * * ? license
#
# В 9:00 с понедельника по пятницу (русские дни недели)
# 0 9 ? * ПН-ПТ delete_users
#
# Каждый день в 8:00 и 20:00
# 0 8,20 * * ? license
#
# Первого числа каждого месяца в полночь
# 0 0 1 * ? delete_users
#
# Каждый понедельник в 6:30
# 30 6 ? * MON license
#
# Последний день каждого месяца в 23:00
# 0 23 L * ? delete_users
#
# Последняя пятница месяца в 18:00
# 0 18 ? * 5L license
#
# Третья суббота месяца в 10:00
# 0 10 ? * 6#3 delete_users
#
# Ближайший рабочий день к 15-му числу каждого месяца в 9:00
# 0 9 15W * ? license
#
# Каждые 30 минут в рабочие дни
# 0/30 * ? * MON-FRI license
#
# Каждую минуту (для тестирования)
# * * * * ? license
#
# ================================================================================
#                          РАБОЧИЕ РАСПИСАНИЯ
# ================================================================================
#
# Добавьте ваши расписания ниже этой линии:

6 0 ? * MON license
30 7 ? * MON delete_users
